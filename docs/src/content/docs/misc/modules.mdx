import {getLatestInstallable} from "@/utils/versions.ts";

export const latestSnapshot = await getLatestInstallable("4.");

Wyck includes a few publications allowing you to pick only the modules your plugin needs. The current runtime targets
**Minecraft 26.3**. If your plugin does not decode existing Minecraft values, it can use `wyck-api` and `wyck-runtime`
without the optional decoder module.


| Artifact                                             | Contents                                                   |
|------------------------------------------------------|------------------------------------------------------------|
| <code>dev.wyck:wyck:{latestSnapshot}</code>          | Complete bundle                                            |
| <code>dev.wyck:wyck-api:{latestSnapshot}</code>      | Public API                                                 |
| <code>dev.wyck:wyck-runtime:{latestSnapshot}</code>  | Minecraft 26.3 runtime; brings in `wyck-api`               |
| <code>dev.wyck:wyck-decoders:{latestSnapshot}</code> | Optional decoders; brings in `wyck-api` and `wyck-runtime` |
| <code>dev.wyck:wyck-vanilla:{latestSnapshot}</code>  | Shaded API and runtime without decoders                    |
